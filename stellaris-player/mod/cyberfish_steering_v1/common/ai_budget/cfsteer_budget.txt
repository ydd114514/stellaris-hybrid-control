# expansion plans without competing colony plans, as in the vanilla entry.

 resource = alloys
 type = expenditure
 category = starbases
 potential = {
  has_edict = cfsteer_expansion_aggressive
  has_ai_expansion_plan = yes
  NOT = { ai_colonize_plans > 0 }
 }

}
